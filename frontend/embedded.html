// Bare-Metal GPIO Pin Toggle on ARM Cortex-M
#define GPIOA_BASE   0x40020000
#define GPIOA_MODER  (*(volatile uint32_t*)(GPIOA_BASE + 0x00))
#define GPIOA_ODR    (*(volatile uint32_t*)(GPIOA_BASE + 0x14))

void init_led() {
    GPIOA_MODER |= (1 << 10); // Configure Pin 5 as General Purpose Output
}
void toggle_led() {
    GPIOA_ODR ^= (1 << 5);    // Atomic XOR toggle on Pin 5
}
